export async function onRequest(context) { // The original destination URL with your token const targetUrl = "http://114.130.57.233:8080/T-Sports/tracks-v1a1/mono.m3u8?token=SkQuhAXZxgBan1"; try { // Fetch the stream from the source server const response = await fetch(targetUrl, { method: 'GET', headers: { // Keeps the content type as an m3u8 playlist 'Accept': 'application/x-mpegURL, video/MP2T', 'User-Agent': context.request.headers.get('User-Agent') || 'Mozilla/5.0' } }); // Return the stream back to the user under your pages.dev domain return new Response(response.body, { status: response.status, headers: { 'Content-Type': 'application/x-mpegURL', 'Access-Control-Allow-Origin': '*', // Allows IPTV players to load it without CORS errors } }); } catch (error) { return new Response("Error fetching the stream", { status: 500 }); } }
